#!/usr/bin/env bash

# 1. Definições de Cores para o Terminal
GREEN='\033[0;32m'
YELLOW='\033[1;33m'
NC='\033[0m' # Sem Cor

echo -e "${YELLOW}[1/3] Limpando caches locais do Godot...${NC}"

# 2. Limpeza de Cache de Importação (Evita que o Claude leia metadados corrompidos/antigos)
# Remove apenas arquivos de propriedades de importação temporários sem quebrar o projeto
if [ -d ".godot/imported" ]; then
    find .godot/imported -name "*.md5" -type f -delete
    echo -e "${GREEN}✓ Cache de importação MD5 limpo.${NC}"
else
    echo "Pasta .godot não encontrada. Certifique-se de estar na raiz do projeto."
fi

echo -e "${YELLOW}[2/3] Injetando variáveis de ambiente de economia de token...${NC}"

# 3. Configurações de Ambiente para o Claude Code
# Desativa telemetria extra, força compactação agressiva e define comportamento não-verbal
export CLAUDE_AUTO_COMPACT=true
export CLAUDE_VERBOSE=false
export CLAUDE_DISABLE_TELEMETRY=true

# 4. Garante que o arquivo CLAUDE.md exista na pasta
if [ ! -f "CLAUDE.md" ]; then
    echo -e "${YELLOW}⚠️ CLAUDE.md não encontrado. Criando um template básico...${NC}"
    echo -e "# Code Style\n- Concise GDScript\n- Static typing\n- No explanations" > CLAUDE.md
fi

echo -e "${YELLOW}[3/3] Iniciando Claude Code com esforço reduzido...${NC}"
echo -e "${GREEN}▶ Claude ativo. Lembre-se de usar /clear ao terminar uma task!${NC}\n"

# 5. Executa o Claude Code forçando o modo de baixo consumo de raciocínio (Low Effort)
# O comando '--effort low' garante que ele gaste menos tokens pensando em tarefas simples
claude --effort low
